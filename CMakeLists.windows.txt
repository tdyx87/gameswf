cmake_minimum_required(VERSION 3.16)
project(gameswf VERSION 1.0.0 LANGUAGES C CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)

# Build options
option(BUILD_SHARED_LIBS "Build shared libraries" OFF)
option(GAMESWF_BUILD_PLAYER "Build gameswf_test_ogl player" ON)
option(GAMESWF_ENABLE_SOUND "Enable sound support via SDL_mixer" OFF)
option(GAMESWF_ENABLE_FREETYPE "Enable FreeType for font rendering" OFF)

# Output directories
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Windows specific settings
if(WIN32)
    # Use vcpkg if available
    if(DEFINED ENV{VCPKG_ROOT})
        set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
    endif()
    
    # Static linking for MSVC
    if(MSVC)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    
    add_definitions(-DWIN32=1 -D_WIN32=1 -D_CRT_SECURE_NO_WARNINGS)
    set(PLATFORM_LIBS ws2_32 winmm opengl32 glu32)
endif()

# Find required packages
find_package(OpenGL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)

# SDL2 - try different methods
find_package(SDL2 QUIET)
if(NOT SDL2_FOUND)
    # Try pkg-config fallback
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(SDL2 QUIET sdl2)
    endif()
endif()

# If still not found, try manual search
if(NOT SDL2_FOUND)
    find_path(SDL2_INCLUDE_DIR SDL.h
        PATHS 
            ${CMAKE_PREFIX_PATH}/include
            $ENV{SDL2_DIR}/include
        PATH_SUFFIXES SDL2
    )
    find_library(SDL2_LIBRARY SDL2
        PATHS 
            ${CMAKE_PREFIX_PATH}/lib
            $ENV{SDL2_DIR}/lib
    )
    if(SDL2_INCLUDE_DIR AND SDL2_LIBRARY)
        set(SDL2_FOUND TRUE)
        set(SDL2_INCLUDE_DIRS ${SDL2_INCLUDE_DIR})
        set(SDL2_LIBRARIES ${SDL2_LIBRARY})
    endif()
endif()

if(NOT SDL2_FOUND)
    message(FATAL_ERROR "SDL2 not found! Please install SDL2 or set SDL2_DIR environment variable.")
endif()

if(GAMESWF_ENABLE_FREETYPE)
    find_package(Freetype QUIET)
    if(FREETYPE_FOUND)
        add_definitions(-DHAVE_FREETYPE=1)
    endif()
endif()

# Platform-specific definitions
if(UNIX)
    add_definitions(-DUNIX=1)
    set(PLATFORM_LIBS ${PLATFORM_LIBS} m dl pthread)
endif()

        /wd4996

# Create base library
add_library(base ${BASE_SOURCES})
target_link_libraries(base
    PUBLIC
        ${SDL2_LIBRARIES}
        ZLIB::ZLIB
        PNG::PNG
        JPEG::JPEG
        ${PLATFORM_LIBS}
)

# Create gameswf library
add_library(gameswf ${GAMESWF_SOURCES})
target_link_libraries(gameswf
    PUBLIC
        base
        OpenGL::GL
        OpenGL::GLU
)

if(FREETYPE_FOUND)
    target_link_libraries(gameswf PUBLIC ${FREETYPE_LIBRARIES})
endif()

# Build the player executable
if(GAMESWF_BUILD_PLAYER)
    add_executable(gameswf_test_ogl gameswf/gameswf_test_ogl.cpp)
    target_link_libraries(gameswf_test_ogl
        PRIVATE
            gameswf
            ${SDL2_LIBRARIES}
    )
    
    # Set Windows subsystem
    if(WIN32)
        set_target_properties(gameswf_test_ogl PROPERTIES
            WIN32_EXECUTABLE FALSE
        )
    endif()
endif()

# Install targets
install(TARGETS base gameswf
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
)

if(GAMESWF_BUILD_PLAYER)
    install(TARGETS gameswf_test_ogl RUNTIME DESTINATION bin)
