cmake_minimum_required(VERSION 3.16)
project(gameswf VERSION 1.0.0 LANGUAGES C CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)

# Build options
option(BUILD_SHARED_LIBS "Build shared libraries" OFF)
option(GAMESWF_BUILD_PLAYER "Build gameswf_test_ogl player" ON)
option(GAMESWF_ENABLE_SOUND "Enable sound support via SDL_mixer" ON)
option(GAMESWF_ENABLE_FREETYPE "Enable FreeType for font rendering" ON)

# Output directories
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Platform-specific definitions
if(WIN32)
    add_definitions(-DWIN32=1 -D_WIN32=1 -D_CRT_SECURE_NO_WARNINGS)
    set(PLATFORM_LIBS ws2_32 winmm opengl32 glu32)
    
    # Static linking for MSVC
    if(MSVC)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
elseif(UNIX)
    add_definitions(-DUNIX=1)
    set(PLATFORM_LIBS m dl pthread)
endif()

# Find required packages (from Conan)
find_package(SDL2 REQUIRED)
find_package(ZLIB REQUIRED)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(OpenGL REQUIRED)

# Find optional packages
if(GAMESWF_ENABLE_FREETYPE)
    find_package(Freetype)
    if(FREETYPE_FOUND)
        add_definitions(-DTU_CONFIG_LINK_TO_FREETYPE=1)
    endif()
endif()

if(GAMESWF_ENABLE_SOUND)
    message(STATUS "GAMESWF_ENABLE_SOUND is ON, looking for SDL_mixer...")
    # Include SDL2_mixer config directly from Conan generators
    set(SDL2_MIXER_CONFIG_PATH "${CMAKE_SOURCE_DIR}/build_conan/build/generators/SDL2_mixerConfig.cmake")
    message(STATUS "Looking for SDL2_mixer config at: ${SDL2_MIXER_CONFIG_PATH}")
    if(EXISTS "${SDL2_MIXER_CONFIG_PATH}")
        include("${SDL2_MIXER_CONFIG_PATH}")
        message(STATUS "SDL2_mixer config file found and included")
    else()
        message(STATUS "SDL2_mixer config file NOT found")
    endif()
    
    if(TARGET SDL2_mixer::SDL2_mixer)
        message(STATUS "SDL_mixer found via SDL2_mixer::SDL2_mixer target")
        set(SDL2_mixer_FOUND TRUE)
        add_definitions(-DGAMESWF_HAVE_SDL_MIXER=1)
    else()
        message(STATUS "SDL_mixer not found, sound disabled")
    endif()
else()
    message(STATUS "GAMESWF_ENABLE_SOUND is OFF")
endif()

# Compiler warnings and flags
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(
        -Wall
        -Wno-unused-variable
        -Wno-unused-but-set-variable
        -Wno-deprecated-declarations
        -Wno-sign-compare
        -Wno-narrowing
        -fPIC
    )
elseif(MSVC)
    add_compile_options(
        /W3
        /wd4244
        /wd4267
        /wd4305
        /wd4996
    )
endif()

# Include directories
include_directories(
    ${CMAKE_SOURCE_DIR}
    ${SDL2_INCLUDE_DIRS}
)

if(FREETYPE_FOUND)
    include_directories(${FREETYPE_INCLUDE_DIRS})
endif()

if(SDL2_mixer_FOUND)
    include_directories(${SDL2_mixer_INCLUDE_DIRS})
endif()

# Add optional sources
if(FREETYPE_FOUND)
    list(APPEND GAMESWF_SOURCES gameswf/gameswf_freetype.cpp)
endif()

if(SDL2_mixer_FOUND)
    list(APPEND GAMESWF_SOURCES gameswf/gameswf_sound_handler_sdl.cpp)
else()
    # Use stub sound handler if no SDL_mixer
    list(APPEND GAMESWF_SOURCES gameswf/gameswf_sound_handler_openal.cpp)
endif()

# Create base library
add_library(base ${BASE_SOURCES})
target_link_libraries(base
    PUBLIC
        SDL2::SDL2
        ZLIB::ZLIB
        PNG::PNG
        JPEG::JPEG
        ${PLATFORM_LIBS}
)

# Create gameswf library
add_library(gameswf ${GAMESWF_SOURCES})
target_link_libraries(gameswf
    PUBLIC
        base
        OpenGL::GL
        OpenGL::GLU
)

if(FREETYPE_FOUND)
    target_link_libraries(gameswf PUBLIC ${FREETYPE_LIBRARIES})
endif()

if(SDL2_mixer_FOUND)
    if(TARGET SDL2_mixer::SDL2_mixer)
        target_link_libraries(gameswf PUBLIC SDL2_mixer::SDL2_mixer)
    elseif(TARGET SDL2_mixer::SDL2_mixer-static)
        target_link_libraries(gameswf PUBLIC SDL2_mixer::SDL2_mixer-static)
    endif()
endif()

# Build the player executable
if(GAMESWF_BUILD_PLAYER)
    add_executable(gameswf_test_ogl gameswf/gameswf_test_ogl.cpp)
    target_link_libraries(gameswf_test_ogl
        PRIVATE
            gameswf
            SDL2::SDL2
    )
    
    # Set Windows subsystem
    if(WIN32)
        set_target_properties(gameswf_test_ogl PROPERTIES
            WIN32_EXECUTABLE FALSE
        )
    endif()
endif()

# Install targets
install(TARGETS base gameswf
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
)

if(GAMESWF_BUILD_PLAYER)
    install(TARGETS gameswf_test_ogl RUNTIME DESTINATION bin)
endif()
